class RepoManager:
    """Gestiona repositorios Git para proyectos."""
    def __init__(self):
        self.repos = CONFIG["repos"]
        self.status_cache = {}

    def _git(self, repo_name: str, *args) -> Dict:
        repo_path = self.repos[repo_name]["path"]
        if not os.path.exists(repo_path):
            return {"error": f"Repositorio {repo_name} no encontrado en {repo_path}"}
        cmd = ["git", "-C", repo_path] + list(args)
        try:
            result = subprocess.run(cmd, capture_output=True, text=True, timeout=30)
            return {
                "stdout": result.stdout.strip(),
                "stderr": result.stderr.strip(),
                "code": result.returncode
            }
        except subprocess.TimeoutExpired:
            return {"error": "Timeout ejecutando git"}
        except Exception as e:
            return {"error": str(e)}

    def get_status(self, repo_name: str) -> Dict:
        status = self._git(repo_name, "status", "--porcelain")
        branch = self._git(repo_name, "branch", "--show-current")
        remote = self._git(repo_name, "remote", "-v")
        return {
            "branch": branch.get("stdout", "unknown"),
            "has_changes": bool(status.get("stdout", "")),
            "changes": status.get("stdout", ""),
            "remote": remote.get("stdout", ""),
            "error": status.get("error") or branch.get("error") or remote.get("error")
        }

    def pull(self, repo_name: str) -> Dict:
        return self._git(repo_name, "pull", self.repos[repo_name]["remote"], self.repos[repo_name]["branch"])

    def push(self, repo_name: str) -> Dict:
        return self._git(repo_name, "push", self.repos[repo_name]["remote"], self.repos[repo_name]["branch"])

    def sync(self, repo_name: str) -> Dict:
        pull_res = self.pull(repo_name)
        if "error" in pull_res:
            return {"pull": pull_res, "push": {"error": "Pull fallido, no se ejecuta push"}}
        push_res = self.push(repo_name)
        return {"pull": pull_res, "push": push_res}

    def get_all_status(self) -> Dict:
        status = {}
        for name in self.repos:
            if self.repos[name].get("enabled", True):
                status[name] = self.get_status(name)
        return status

            return False

        if self.allowed_users and user_id not in self.allowed_users:
            return "⛔ Usuario no autorizado. Contacta al administrador."
        if cmd in self.commands:
            return self.commands[cmd](args)
        return f"❌ Comando desconocido. Usa /help para ver los disponibles."

class SystemMonitor(threading.Thread):
    """Monitor de recursos del sistema y alertas."""
    def __init__(self, bot: TelegramBot, interval: int = 30):
        super().__init__()
        self.bot = bot
        self.interval = interval
        self.daemon = True
        self.alert_cpu = False
        self.alert_ram = False
        self.running = True
        logger.info("🛡️ System Monitor iniciado.")

    def run(self):
        while self.running:
            try:
                cpu = psutil.cpu_percent(interval=1)
                mem = psutil.virtual_memory().percent
                # Alertas CPU
                if cpu > CONFIG["monitor"]["cpu_threshold"]:
                    if not self.alert_cpu:
                        self.bot.send_message(
                            f"⚠️ **ALERTA CPU**: Uso elevado al {cpu}%. Verificar procesos con /top_processes."
                        )
                        self.alert_cpu = True
                else:
                    self.alert_cpu = False
                # Alertas RAM
                if mem > CONFIG["monitor"]["ram_threshold"]:
                    if not self.alert_ram:
                        self.bot.send_message(
                            f"⚠️ **ALERTA RAM**: Uso elevado al {mem}%. Posible fuga de memoria."
                        )
                        self.alert_ram = True
                else:
                    self.alert_ram = False
                # Verificar que la API responde
                try:
                    requests.get(f"http://localhost:{CONFIG['api']['port']}/health", timeout=3)
                except:
                    logger.warning("La API no responde. Verificar proceso.")
                    self.bot.send_message("⚠️ **ALERTA**: El servidor API no responde. Verificar el proceso.")
            except Exception as e:
                logger.error(f"Error en monitor: {e}")
            time.sleep(self.interval)

    def stop(self):
        self.running = False

SNAPSHOT_DIR = os.path.expanduser("~/.orchestrator/snapshots")
os.makedirs(SNAPSHOT_DIR, exist_ok=True)

@app.post("/api/snapshot/capture")
async def capture_snapshot(request: Request, auth=Depends(verify_token)):
    data = await request.json()
    case_id = data.get("case_id", f"SNAP-{int(time.time())}")
    case_dir = Path(SNAPSHOT_DIR) / case_id
    case_dir.mkdir(parents=True, exist_ok=True)
    
    # Captura de información del sistema (no forense intrusiva)
    snapshots = {}
    snapshots["processes"] = subprocess.run("ps aux", shell=True, capture_output=True, text=True).stdout
    snapshots["network"] = subprocess.run("netstat -tunap 2>/dev/null || ss -tunap", shell=True, capture_output=True, text=True).stdout
    snapshots["system_info"] = subprocess.run("uname -a", shell=True, capture_output=True, text=True).stdout
    snapshots["disk_usage"] = subprocess.run("df -h", shell=True, capture_output=True, text=True).stdout
    
    import hashlib
    manifest = {}
    for name, content in snapshots.items():
        (case_dir / f"{name}.txt").write_text(content)
        manifest[name] = {
            "sha256": hashlib.sha256(content.encode()).hexdigest(),
            "size": len(content)
        }
    (case_dir / "manifest.json").write_text(json.dumps(manifest, indent=2))
    
    import zipfile
    zip_path = case_dir / f"{case_id}.zip"
    with zipfile.ZipFile(zip_path, 'w', zipfile.ZIP_DEFLATED) as zf:
        for f in case_dir.glob("*.txt"):
            zf.write(f, f.name)
    return {"case_id": case_id, "zip": str(zip_path), "files": len(snapshots)}

```bash
scan_network() {
    local target="${1:-192.168.1.0/24}"
    echo "🔍 Escaneando red $target..."
    nmap -sn "$target" | grep "Nmap scan" | cut -d' ' -f5
}

# ============================================================
# 2. ESCANEO DE PUERTOS (rápido)
# ============================================================
scan_ports() {
    local target="$1"
    local ports="${2:-22,80,443,554,8080,8443,3306,5432,6379}"
    echo "🔌 Escaneando puertos en $target..."
    nmap -p "$ports" --open "$target" | grep -E "open|closed"
}

# ============================================================

# ============================================================
scan_cameras() {
    local target="${1:-192.168.1.0/24}"

    nmap -p 554,80,8080,37777 --open "$target" -oG - | grep -E "554/open|80/open"
}

    echo "📱 Escaneando Bluetooth..."

        hcitool scan
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
COM-LINK CRYPTO — Cifrado de extremo a extremo
Uso: ./crypto.py --encrypt "mensaje" --key "clave"
"""

import os
import sys
import json
import base64
import hashlib
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
import argparse

# ============================================================
# GENERACIÓN DE CLAVE
# ============================================================
def generate_key(password: str, salt: bytes = None) -> bytes:
    if salt is None:
        salt = os.urandom(16)
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=32,
        salt=salt,
        iterations=100000,
    )
    key = base64.urlsafe_b64encode(kdf.derive(password.encode()))
    return key

# ============================================================
# CIFRADO AES-GCM (Más seguro)
# ============================================================
def encrypt_aes_gcm(plaintext: str, password: str) -> str:
    salt = os.urandom(16)
    key = generate_key(password, salt)
    aesgcm = AESGCM(key)
    nonce = os.urandom(12)
    ciphertext = aesgcm.encrypt(nonce, plaintext.encode(), None)
    return base64.b64encode(salt + nonce + ciphertext).decode()

def decrypt_aes_gcm(ciphertext_b64: str, password: str) -> str:
    data = base64.b64decode(ciphertext_b64)
    salt = data[:16]
    nonce = data[16:28]
    ciphertext = data[28:]
    key = generate_key(password, salt)
    aesgcm = AESGCM(key)
    plaintext = aesgcm.decrypt(nonce, ciphertext, None)
    return plaintext.decode()

# ============================================================
# CIFRADO CON FERNET (Más simple)
# ============================================================
def encrypt_fernet(plaintext: str, password: str) -> str:
    key = generate_key(password)
    f = Fernet(key)
    return f.encrypt(plaintext.encode()).decode()

def decrypt_fernet(ciphertext: str, password: str) -> str:
    key = generate_key(password)
    f = Fernet(key)
    return f.decrypt(ciphertext.encode()).decode()

# ============================================================
# GENERACIÓN DE HASH
# ============================================================
def hash_message(message: str) -> str:
    return hashlib.sha256(message.encode()).hexdigest()

# ============================================================
# CLI
# ============================================================
def main():
    parser = argparse.ArgumentParser(description="COM-LINK Crypto")
    parser.add_argument("--encrypt", help="Texto a cifrar")
    parser.add_argument("--decrypt", help="Texto a descifrar")
    parser.add_argument("--key", help="Clave de cifrado", required=True)
    parser.add_argument("--hash", help="Generar hash del mensaje")
    parser.add_argument("--method", default="fernet", choices=["fernet", "aes-gcm"], help="Método de cifrado")
    
    args = parser.parse_args()
    
    if args.encrypt:
        if args.method == "fernet":
            result = encrypt_fernet(args.encrypt, args.key)
        else:
            result = encrypt_aes_gcm(args.encrypt, args.key)
        print(f"🔐 Cifrado: {result}")
    
    elif args.decrypt:
        if args.method == "fernet":
            result = decrypt_fernet(args.decrypt, args.key)
        else:
            result = decrypt_aes_gcm(args.decrypt, args.key)
        print(f"🔓 Descifrado: {result}")
    
    elif args.hash:
        result = hash_message(args.hash)
        print(f"🔑 Hash: {result}")
    
    else:
        print("Uso: crypto.py --encrypt 'mensaje' --key 'clave'")

if __name__ == "__main__":
    main()
```

```bash
                ;;

                ./channel_manager.sh --all "PONG"
                ;;
        esac
```

```bash
    if [ -n "$TELEGRAM_BOT_TOKEN" ] && [ -n "$TELEGRAM_CHAT_ID" ]; then
        echo "🟢 Activo"
    else
        echo "🔴 No configurado"
    fi
}

check_sms() {
    if command -v termux-sms-send >/dev/null 2>&1; then
        echo "🟢 Disponible"
    else
        echo "🔴 No disponible"
    fi
}

check_sip() {
    if [ -f "$CONFIG_DIR/linphonerc" ] && [ -n "$SIP_USER" ]; then
        echo "🟢 Configurado"
    else
        echo "🔴 No configurado"
    fi
}

check_mesh() {
    if command -v meshtastic >/dev/null 2>&1; then
        echo "🟢 Disponible"
    else
        echo "🔴 No instalado"
    fi
}

check_radio() {
    if [ -c "/dev/ttyUSB0" ] || [ -c "/dev/ttyACM0" ]; then
        echo "🟢 Conectado"
    else
        echo "🔴 Desconectado"
    fi
}

check_wifi() {
    if command -v termux-wifi-scan >/dev/null 2>&1; then
        echo "🟢 Activo"
    else
        echo "🔴 Inactivo"
    fi
}

check_battery() {
    if command -v termux-battery-status >/dev/null 2>&1; then
        termux-battery-status | jq -r '.percentage'
    else
        echo "N/A"
    fi
}

check_gps() {
    if command -v termux-location >/dev/null 2>&1; then
        echo "🟢 Disponible"
    else
        echo "🔴 No disponible"
    fi
}

# ============================================================
# DASHBOARD
# ============================================================
dashboard() {
    clear
    echo -e "${BLUE}╔══════════════════════════════════════════════════════╗${NC}"
    echo -e "${BLUE}║  📡 COM-LINK DASHBOARD — SISTEMA OPERATIVO        ║${NC}"
    echo -e "${BLUE}╠══════════════════════════════════════════════════════╣${NC}"
    echo -e "${BLUE}║  Estado del sistema:                               ║${NC}"
    echo -e "${BLUE}║    Batería: $(check_battery)%                            ║${NC}"
    echo -e "${BLUE}║    GPS:     $(check_gps)                            ║${NC}"
    echo -e "${BLUE}║    WiFi:    $(check_wifi)                            ║${NC}"
    echo -e "${BLUE}╠══════════════════════════════════════════════════════╣${NC}"
    echo -e "${BLUE}║  Canales:                                          ║${NC}"
    echo -e "${BLUE}║    Telegram: $(check_telegram)                     ║${NC}"
    echo -e "${BLUE}║    SMS:      $(check_sms)                     ║${NC}"
    echo -e "${BLUE}║    SIP:      $(check_sip)                     ║${NC}"
    echo -e "${BLUE}║    Malla:    $(check_mesh)                     ║${NC}"
    echo -e "${BLUE}║    Radio:    $(check_radio)                     ║${NC}"
    echo -e "${BLUE}╠══════════════════════════════════════════════════════╣${NC}"
    echo -e "${BLUE}║  Últimas alertas:                                  ║${NC}"
    tail -n 3 ~/.com_link/logs/transmissions.log 2>/dev/null || echo "   (Sin eventos)"
    echo -e "${BLUE}╚══════════════════════════════════════════════════════╝${NC}"
}

# ============================================================
# CONTROL
# ============================================================
control() {
    echo -e "${BLUE}┌────────────────────────────────────────────────────┐${NC}"
    echo -e "${BLUE}│  🎮 COM-LINK CONTROL CENTER                     │${NC}"
    echo -e "${BLUE}├────────────────────────────────────────────────────┤${NC}"
    echo "1.  Enviar mensaje a todos los canales"
    echo "2.  Enviar solo a Telegram"
    echo "3.  Enviar SMS"
    echo "4.  Llamada SIP"
    echo "5.  Activar WiFi Ad-hoc"
    echo "6.  Escanear red"
    echo "7.  Enviar ubicación"
    echo "8.  Alerta de emergencia (PANIC)"
    echo "9.  Ver logs"
    echo "10. Configuración"
    echo "0.  Salir"
    echo -e "${BLUE}└────────────────────────────────────────────────────┘${NC}"
    read -p "👉 Selecciona: " choice
    
    case $choice in
        1) read -p "💬 Mensaje: " msg; ./channel_manager.sh --all "$msg" ;;
        2) read -p "💬 Mensaje: " msg; ./channel_manager.sh --telegram "$msg" ;;
        3) read -p "📱 Número: " num; read -p "💬 Mensaje: " msg; ./channel_manager.sh --sms "$num" "$msg" ;;
        4) read -p "📞 Número: " num; read -p "💬 Mensaje: " msg; ./channel_manager.sh --sip "$num" "$msg" ;;
        5) ./wifi_ad_hoc.sh ;;
        6) ./scanner_universal.sh ;;
        7) ./get_location.sh ;;
        8) ./panic_alert.sh ;;
        9) tail -f ~/.com_link/logs/transmissions.log ;;
        10) nano ~/.com_link/config/.env ;;
        0) exit 0 ;;
        *) echo -e "${RED}❌ Opción inválida${NC}" ;;
    esac
}

# ============================================================
# MAIN
# ============================================================
if [ $# -eq 0 ]; then
    dashboard
    echo ""
    control
else
    case "$1" in
        --status) dashboard ;;
        --help)
            echo "COM-LINK DASHBOARD"
            echo "  --status   Mostrar estado"
            echo "  --control  Mostrar control"
            ;;
        *) echo "❌ Opción desconocida" ;;
    esac
fi
```

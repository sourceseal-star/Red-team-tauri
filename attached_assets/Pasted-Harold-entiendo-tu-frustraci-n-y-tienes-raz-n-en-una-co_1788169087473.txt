## 📄 ARCHIVO 1: `watcher.py` (Hot-reload automático)

Guárdalo en `~/Red-team-tauri/watcher.py`:

```python
#!/data/data/com.termux/files/usr/bin/python
# -*- coding: utf-8 -*-
"""
WATCHER — Recarga automática de módulos cuando cambian.
Monitorea redteam/modules/ y registra cambios en ledger SourceSeal.
Uso: python3 watcher.py [--dashboard-pid PID]
"""
import os, sys, time, json, hashlib, subprocess, signal
from pathlib import Path
from datetime import datetime

MODULES_DIR = Path(__file__).parent / "redteam" / "modules"
LEDGER = Path.home() / ".c2" / "evidence_ledger.json"
STATE_FILE = Path.home() / ".c2" / "module_state.json"
LOG = Path.home() / ".c2" / "logs" / "watcher.log"

LOG.parent.mkdir(parents=True, exist_ok=True)

def log(msg):
    ts = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    line = f"[{ts}] {msg}"
    print(line)
    with open(LOG, "a") as f:
        f.write(line + "\n")

def file_hash(path: Path) -> str:
    if not path.exists():
        return ""
    return hashlib.sha256(path.read_bytes()).hexdigest()

def load_state() -> dict:
    if STATE_FILE.exists():
        return json.loads(STATE_FILE.read_text())
    return {}

def save_state(state: dict):
    STATE_FILE.write_text(json.dumps(state, indent=2))

def seal_module_change(module_name: str, new_hash: str, change_type: str):
    """Registra el cambio en el ledger SourceSeal."""
    if not LEDGER.exists():
        return
    try:
        ledger = json.loads(LEDGER.read_text())
        prev = ledger["chain_hash"]
        payload = f"{prev}|module:{module_name}|{change_type}|{new_hash}|{int(time.time())}"
        new_chain = hashlib.sha256(payload.encode()).hexdigest()
        ledger["events"].append({
            "type": "MODULE_CHANGE",
            "module": module_name,
            "change": change_type,
            "sha256": new_hash,
            "chain_hash": new_chain,
            "timestamp": datetime.utcnow().isoformat() + "Z"
        })
        ledger["chain_hash"] = new_chain
        if len(ledger["events"]) > 1000:
            ledger["events"] = ledger["events"][-1000:]
        LEDGER.write_text(json.dumps(ledger, indent=2))
        log(f"🔗 Sellado: {module_name} ({change_type})")
    except Exception as e:
        log(f"⚠️ Error sellando: {e}")

def signal_dashboard(action: str = "reload"):
    """Envía señal al dashboard para que recargue módulos."""
    pid_file = Path.home() / ".c2" / "pids" / "dashboard.pid"
    if pid_file.exists():
        try:
            pid = int(pid_file.read_text().strip())
            sig = signal.SIGUSR1 if action == "reload" else signal.SIGTERM
            os.kill(pid, sig)
            log(f"📡 Señal {action} enviada a dashboard (PID {pid})")
        except Exception as e:
            log(f"⚠️ No se pudo señalizar dashboard: {e}")

def scan_modules() -> dict:
    """Escanea todos los .py en modules/."""
    state = {}
    if not MODULES_DIR.exists():
        return state
    for f in MODULES_DIR.glob("*.py"):
        if f.name.startswith("_"):
            continue
        state[f.stem] = {
            "hash": file_hash(f),
            "size": f.stat().st_size,
            "mtime": f.stat().st_mtime
        }
    return state

def main():
    log("=" * 60)
    log("👁️ WATCHER iniciado — monitoreando módulos Red Team")
    log(f"   Directorio: {MODULES_DIR}")
    log(f"   Ledger:     {LEDGER}")
    log("=" * 60)

    previous = load_state()
    if not previous:
        previous = scan_modules()
        save_state(previous)
        log(f"📦 Estado inicial: {len(previous)} módulos catalogados")
        for name in previous:
            seal_module_change(name, previous[name]["hash"], "BASELINE")

    try:
        while True:
            current = scan_modules()
            changes = []

            # Módulos nuevos
            for name in current:
                if name not in previous:
                    changes.append(("ADDED", name, current[name]["hash"]))

            # Módulos eliminados
            for name in previous:
                if name not in current:
                    changes.append(("REMOVED", name, ""))

            # Módulos modificados
            for name in current:
                if name in previous and current[name]["hash"] != previous[name]["hash"]:
                    changes.append(("MODIFIED", name, current[name]["hash"]))

            if changes:
                for change_type, name, h in changes:
                    icon = {"ADDED": "➕", "REMOVED": "➖", "MODIFIED": "🔄"}[change_type]
                    log(f"{icon} {change_type}: {name}.py ({h[:12]})")
                    seal_module_change(name, h, change_type)

                save_state(current)
                signal_dashboard("reload")
                previous = current

            time.sleep(2)

    except KeyboardInterrupt:
        log("🛑 Watcher detenido")

if __name__ == "__main__":
    main()
```

```python
# -*- coding: utf-8 -*-
"""
HOT LOADER — Carga y recarga módulos de redteam/modules/ bajo demanda.
Permite al dashboard usar versiones actualizadas sin reiniciar.
"""
import importlib, importlib.util, sys, signal
from pathlib import Path
from typing import Optional, Dict, Any

MODULES_DIR = Path(__file__).parent.parent / "modules"
_loaded: Dict[str, Any] = {}
_mtimes: Dict[str, float] = {}

def _reload_on_signal(signum, frame):
    """Handler SIGUSR1: recarga todos los módulos."""
    print("[HOT-LOADER] Recargando módulos...")
    reload_all()

# Registrar handler
try:
    signal.signal(signal.SIGUSR1, _reload_on_signal)
except AttributeError:
    pass  # Windows no tiene SIGUSR1

def load(module_name: str, force: bool = False):
    """Carga un módulo. Si ya estaba cargado y cambió, lo recarga."""
    path = MODULES_DIR / f"{module_name}.py"
    if not path.exists():
        raise ImportError(f"Módulo no encontrado: {module_name}")

    mtime = path.stat().st_mtime

    if module_name in _loaded and not force:
        if _mtimes.get(module_name) == mtime:
            return _loaded[module_name]

    spec = importlib.util.spec_from_file_location(
        f"redteam.modules.{module_name}", str(path)
    )
    mod = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = mod
    spec.loader.exec_module(mod)
    _loaded[module_name] = mod
    _mtimes[module_name] = mtime
    print(f"[HOT-LOADER] Cargado: {module_name}.py (mtime={mtime})")
    return mod

def reload_all():
    """Recarga todos los módulos conocidos."""
    for name in list(_loaded.keys()):
        try:
            load(name, force=True)
        except Exception as e:
            print(f"[HOT-LOADER] Error recargando {name}: {e}")

def available() -> list:
    """Lista módulos disponibles."""
    if not MODULES_DIR.exists():
        return []
    return [f.stem for f in MODULES_DIR.glob("*.py")
            if not f.name.startswith("_")]
```

## 📄 ARCHIVO 3: `redteam/runner/engagement_guard.py`

```python
# -*- coding: utf-8 -*-
"""
ENGAGEMENT GUARD — Valida que un objetivo esté dentro del alcance autorizado.
Bloquea cualquier acción fuera de CIDRs/hosts permitidos.
"""
import json, ipaddress, re
from pathlib import Path
from datetime import datetime
from typing import Optional

ENGAGEMENTS_FILE = Path.home() / "Red-team-tauri" / "config" / "engagements.json"
AUDIT_LOG = Path.home() / ".c2" / "logs" / "engagement_audit.log"

def _load_engagements() -> dict:
    if not ENGAGEMENTS_FILE.exists():
        return {"engagements": []}
    return json.loads(ENGAGEMENTS_FILE.read_text())

def _audit(action: str, target: str, allowed: bool, reason: str):
    AUDIT_LOG.parent.mkdir(parents=True, exist_ok=True)
    entry = f"[{datetime.utcnow().isoformat()}Z] {action} | {target} | {'✅' if allowed else '❌'} | {reason}\n"
    with open(AUDIT_LOG, "a") as f:
        f.write(entry)

def is_target_authorized(target: str, engagement_id: Optional[str] = None) -> bool:
    """Verifica si target (IP, CIDR o dominio) está en alcance autorizado."""
    data = _load_engagements()
    now = datetime.utcnow()

    for eng in data.get("engagements", []):
        if engagement_id and eng.get("id") != engagement_id:
            continue

        # Verificar vigencia
        start = datetime.fromisoformat(eng["start_date"])
        end = datetime.fromisoformat(eng["end_date"])
        if not (start <= now <= end):
            continue

        # Verificar firma de autorización
        if not eng.get("authorization_signed"):
            continue

        for scope in eng.get("scope", []):
            if _matches(target, scope):
                _audit("AUTH_CHECK", target, True, eng["id"])
                return True

    _audit("AUTH_CHECK", target, False, "fuera de alcance")
    return False

def _matches(target: str, scope_entry: str) -> bool:
    """Match IP, CIDR o dominio."""
    # CIDR
    if "/" in scope_entry:
        try:
            net = ipaddress.ip_network(scope_entry, strict=False)
            ip = ipaddress.ip_address(target)
            return ip in net
        except ValueError:
            pass

    # IP exacta
    try:
        ipaddress.ip_address(scope_entry)
        ipaddress.ip_address(target)
        return target == scope_entry
    except ValueError:
        pass

    # Dominio (soporta wildcards: *.example.com)
    if scope_entry.startswith("*."):
        pattern = r".*\." + re.escape(scope_entry[2:]) + r"$"
        return bool(re.match(pattern, target))
    return target == scope_entry

def get_active_engagement(engagement_id: str) -> Optional[dict]:
    data = _load_engagements()
    for eng in data.get("engagements", []):
        if eng.get("id") == engagement_id:
            return eng
    return None

def require_authorization(target: str, engagement_id: str) -> bool:
    """Lanza excepción si el target no está autorizado."""
    eng = get_active_engagement(engagement_id)
    if not eng:
        raise PermissionError(f"Engagement {engagement_id} no existe")
    if not eng.get("authorization_signed"):
        raise PermissionError(f"Engagement {engagement_id} sin autorización firmada")
    if not is_target_authorized(target, engagement_id):
        raise PermissionError(f"Target {target} fuera de alcance autorizado")
    return True
```

## 📄 ARCHIVO 4: `redteam/modules/base.py` (clase base)

```python
# -*- coding: utf-8 -*-
"""
Clase base para todos los módulos de red team.
Implementa: validación de alcance, evidencia firmada, auditoría.
"""
import json, hashlib, time
from pathlib import Path
from datetime import datetime
from typing import Dict, Any, Optional
from redteam.runner.engagement_guard import require_authorization

EVIDENCE_DIR = Path.home() / "Red-team-tauri" / "evidence" / "findings"
EVIDENCE_DIR.mkdir(parents=True, exist_ok=True)

class BaseModule:
    name: str = "base"
    description: str = ""
    version: str = "1.0"
    risk_level: str = "low"  # low | medium | high | critical

    def __init__(self, engagement_id: str):
        self.engagement_id = engagement_id
        self.start_time = None
        self.findings = []

    def authorize_target(self, target: str) -> bool:
        """Valida target contra el alcance del engagement."""
        return require_authorization(target, self.engagement_id)

    def run(self, target: str, **kwargs) -> Dict[str, Any]:
        """Punto de entrada principal. Subclases sobrescriben _execute."""
        self.start_time = datetime.utcnow()
        if not self.authorize_target(target):
            return {"error": "target_no_autorizado", "target": target}
        try:
            result = self._execute(target, **kwargs)
            self._seal_finding(target, result)
            return result
        except Exception as e:
            return {"error": str(e), "module": self.name}

    def _execute(self, target: str, **kwargs) -> Dict[str, Any]:
        raise NotImplementedError

    def _seal_finding(self, target: str, result: Dict):
        """Guarda hallazgo con hash encadenado (mini SourceSeal)."""
        finding = {
            "module": self.name,
            "version": self.version,
            "engagement_id": self.engagement_id,
            "target": target,
            "timestamp": datetime.utcnow().isoformat() + "Z",
            "duration_sec": (datetime.utcnow() - self.start_time).total_seconds(),
            "result": result,
        }
        content = json.dumps(finding, sort_keys=True)
        finding_hash = hashlib.sha256(content.encode()).hexdigest()
        finding["finding_hash"] = finding_hash

        out_file = EVIDENCE_DIR / f"{self.name}_{int(time.time())}_{finding_hash[:8]}.json"
        out_file.write_text(json.dumps(finding, indent=2))
        self.findings.append(finding)
```
